   216→/* ---- Counter Component ---- */
   217→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   218→  const ref = useRef<HTMLSpanElement>(null);
   219→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   220→  const [count, setCount] = useState(0);
   221→
   222→  useEffect(() => {
   223→    if (!isInView) return;
   224→    let start = 0;
   225→    const duration = 2000;
   226→    const increment = target / (duration / 16);
   227→    const timer = setInterval(() => {
   228→      start += increment;
   229→      if (start >= target) {
   230→        setCount(target);
   231→        clearInterval(timer);
   232→      } else {
   233→        setCount(Math.floor(start));
   234→      }
   235→    }, 16);
   236→    return () => clearInterval(timer);
   237→  }, [isInView, target]);
   238→
   239→  return (
   240→    <span ref={ref}>
   241→      {count.toLocaleString('de-DE')}
   242→      {suffix}
   243→    </span>
   244→  );
   245→}
