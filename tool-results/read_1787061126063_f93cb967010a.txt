   192→   192→/* ============================================================ */
   193→   193→function AnimatedCounter({ target, suffix }: { target: number; suffix: string }) {
   194→   194→  const ref = useRef<HTMLSpanElement>(null);
   195→   195→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   196→   196→  const [count, setCount] = useState(0);
   197→   197→
   198→   198→  useEffect(() => {
   199→   199→    if (!isInView) return;
   200→   200→    let start = 0;
   201→   201→    const duration = 2000;
   202→   202→    const increment = target / (duration / 16);
   203→   203→    const timer = setInterval(() => {
   204→   204→      start += increment;
   205→   205→      if (start >= target) {
   206→   206→        setCount(target);
   207→   207→        clearInterval(timer);
   208→   208→      } else {
   209→   209→        setCount(Math.floor(start));
   210→   210→      }
   211→   211→    }, 16);
   212→   212→    return () => clearInterval(timer);
   213→   213→  }, [isInView, target]);
   214→   214→
   215→   215→  return (
   216→   216→    <span ref={ref}>
   217→   217→      {count.toLocaleString('de-DE')}
   218→   218→      {suffix}
   219→   219→    </span>
   220→   220→  );
   221→   221→}
